DB_NAME = 'colosseum_accounting.db'

def get_db_connection():
    conn = sqlite3.connect(DB_NAME)
    conn.row_factory = sqlite3.Row
    return conn

def init_db():
    conn = get_db_connection()
    cursor = conn.cursor()
    cursor.execute("""
        CREATE TABLE IF NOT EXISTS transactions (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            date TEXT NOT NULL,
            type TEXT NOT NULL,
            category TEXT NOT NULL,
            amount REAL NOT NULL,
            description TEXT NOT NULL
        )
    """)
    conn.commit()
    
    # Check if database is empty or has old small values; if empty, populate initial LKR sample data starting Jan 2026
    cursor.execute("SELECT COUNT(*) FROM transactions")
    if cursor.fetchone()[0] == 0:
        seed_sample_data(conn)
    
    conn.close()

# CURRENCY CONVERSION HELPERS
def convert_val(amount_lkr, mode, rate):
    """Convert LKR amount to target currency mode."""
    if mode == "USD ($)":
        return amount_lkr / rate if rate > 0 else 0.0
    return amount_lkr

def fmt_curr(amount_lkr, mode, rate, show_code=True):
    """Format LKR amount as string in selected currency."""
    val = convert_val(amount_lkr, mode, rate)
    if mode == "USD ($)":
        return f"${val:,.2f}"
    else:
        return f"Rs. {val:,.2f}" if show_code else f"{val:,.2f}"

        {"code": "5040", "name": "Marketing & Digital Promotion", "type": "Expense", "normal": "Debit", "desc": "Social media promotions and gym flyers."},
        {"code": "5050", "name": "Software, Subscriptions & Music", "type": "Expense", "normal": "Debit", "desc": "Check-in software, camera storage, Spotify music."},
        {"code": "5090", "name": "Miscellaneous Gym Operations", "type": "Expense", "normal": "Debit", "desc": "First aid supplies, disinfectants, ice bags."},
    ]
    coa_dict = {a["code"]: a for a in COA}

    # Generate Journal Entries
    journal_entries = []

    # Opening Entry
    journal_entries.append({
        "entry_num": "JRN-2026-000",
        "date": "2026-01-01",
        "narration": "Establishment of Colosseum Fight Club Opening Balance Sheet: Owner capital injection to fund bank account, cash register float, boxing ring, and gear inventory.",
        "lines": [
            {"code": "1010", "name": "Cash on Hand (Gym Register)", "debit": 150000.0, "credit": 0.0},
            {"code": "1020", "name": "Commercial Bank Current Account", "debit": 1350000.0, "credit": 0.0},

    # Transaction journal lines
    for idx, row in df_raw.iterrows():

                "ref": entry["entry_num"],
                "counterparty": cp,
                "narration": entry["narration"],
                "debit": dr,
                "credit": cr,
                "balance": bal

    for c in ledgers:
        norm = coa_dict[c]["normal"]
        ledgers[c]["closing"] = (ledgers[c]["debits"] - ledgers[c]["credits"]) if norm == "Debit" else (ledgers[c]["credits"] - ledgers[c]["debits"])

        "5. 🏛️ Balance Sheet",
        "6. 📚 Chart of Accounts Guide"

        for entry in journal_entries:

                continue

            with st.expander(f"{entry['entry_num']} • {entry['date']} • {entry['narration'][:70]}...", expanded=True):
                st.caption(f"**Narration:** {entry['narration']}")
                j_rows = []
                for l in entry["lines"]:
                    indent = "    " if l["credit"] > 0 else ""
                    dr_s = fmt_curr(l["debit"], currency_mode, exchange_rate) if l["debit"] > 0 else "—"
                    cr_s = fmt_curr(l["credit"], currency_mode, exchange_rate) if l["credit"] > 0 else "—"
                    j_rows.append({
                        "Account Code": l["code"],
                        "Account Title": indent + ("To " + l["name"] if l["credit"] > 0 else l["name"]),
                        f"Debit ({currency_mode})": dr_s,
                        f"Credit ({currency_mode})": cr_s
                    })
                st.dataframe(pd.DataFrame(j_rows), use_container_width=True, hide_index=True)

            "Select Account Ledger",
            [f"{a['code']} - {a['name']}" for a in COA],

                    f"Credit (Cr)": fmt_curr(p["credit"], currency_mode, exchange_rate) if p["credit"] > 0 else "—",
                    f"Running Balance": fmt_curr(p["balance"], currency_mode, exchange_rate)
                })
            st.dataframe(pd.DataFrame(table_postings), use_container_width=True, hide_index=True)

    # 3. CLASSIC T-ACCOUNTS
    with de_tabs[2]:
        st.markdown("### ⚖️ Classic T-Account Visualization")
        st.caption("Visual Left (Debits) vs Right (Credits) ledger card.")

        net_inc = tot_rev - tot_exp
        tot_equity = cap + net_inc - draw
        tot_liab_equity = tot_liab + tot_equity

        col_bs1, col_bs2 = st.columns(2)
